cmake_minimum_required(VERSION 3.22)
project(jianxin_rime LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(ANDROID)
    # 词库部署与查询不应因测试 APK 的 Debug 标记退化为未优化的 native 代码。
    # 保留调试符号，同时让调试安装包也使用接近正式版的引擎性能。
    add_compile_options($<$<CONFIG:Debug>:-O2>)
endif()
set(RIME_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../third_party/librime")
set(BOOST_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../third_party/boost")
set(Boost_NO_SYSTEM_PATHS ON)
set(Boost_INCLUDE_DIR "${BOOST_ROOT}" CACHE PATH "" FORCE)
include_directories("${BOOST_ROOT}")

# 第三方依赖全部静态链接到自己的 JNI 库；不引入 CyIME 的应用层。
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_STATIC ON CACHE BOOL "" FORCE)
set(BUILD_TEST OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(ENABLE_LOGGING OFF CACHE BOOL "" FORCE)
set(ENABLE_TIMESTAMP OFF CACHE BOOL "" FORCE)
set(YAML_CPP_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(YAML_CPP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(YAML_CPP_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${RIME_ROOT}/deps/yaml-cpp" yaml-cpp)
set(YamlCpp_INCLUDE_PATH "${RIME_ROOT}/deps/yaml-cpp/include" CACHE PATH "" FORCE)
set(YamlCpp_LIBRARY yaml-cpp CACHE STRING "" FORCE)
set(YamlCpp_NEW_API TRUE CACHE BOOL "" FORCE)

set(LEVELDB_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LEVELDB_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(LEVELDB_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${RIME_ROOT}/deps/leveldb" leveldb)
set(LevelDb_INCLUDE_PATH "${RIME_ROOT}/deps/leveldb/include" CACHE PATH "" FORCE)
set(LevelDb_LIBRARY leveldb CACHE STRING "" FORCE)

set(ENABLE_TOOLS OFF CACHE BOOL "" FORCE)
add_subdirectory("${RIME_ROOT}/deps/marisa-trie" marisa)
set(Marisa_INCLUDE_PATH "${RIME_ROOT}/deps/marisa-trie/include" CACHE PATH "" FORCE)
set(Marisa_LIBRARY marisa CACHE STRING "" FORCE)
set(LIBMARISA marisa)
set(USE_SYSTEM_MARISA ON CACHE BOOL "" FORCE)
set(BUILD_DOCUMENTATION OFF CACHE BOOL "" FORCE)
add_subdirectory("${RIME_ROOT}/deps/opencc" opencc EXCLUDE_FROM_ALL)
target_include_directories(libopencc PUBLIC "${CMAKE_CURRENT_BINARY_DIR}/opencc/src")
# RIME 引用 <opencc/Config.hpp>；在构建目录提供安装后的头文件布局。
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/include/opencc")
file(COPY "${RIME_ROOT}/deps/opencc/src/" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/include/opencc"
     FILES_MATCHING PATTERN "*.h" PATTERN "*.hpp")
configure_file("${CMAKE_CURRENT_BINARY_DIR}/opencc/src/Opencc_Export.h"
               "${CMAKE_CURRENT_BINARY_DIR}/include/opencc/Opencc_Export.h" COPYONLY)
set(Opencc_INCLUDE_PATH "${CMAKE_CURRENT_BINARY_DIR}/include" CACHE PATH "" FORCE)
set(Opencc_LIBRARY libopencc CACHE STRING "" FORCE)
set(X11Keysym "${RIME_ROOT}/include" CACHE PATH "" FORCE)

add_subdirectory("${RIME_ROOT}" rime)
if(ANDROID)
add_library(jianxin_rime SHARED rime_bridge.cpp)
target_include_directories(jianxin_rime PRIVATE "${RIME_ROOT}/src" "${RIME_ROOT}/include")
target_link_libraries(jianxin_rime rime-static log)
target_compile_options(jianxin_rime PRIVATE -Wall -Wextra)
target_link_options(jianxin_rime PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
else()
add_executable(jianxin_engine_check engine_check.cpp)
target_include_directories(jianxin_engine_check PRIVATE "${RIME_ROOT}/src")
target_link_libraries(jianxin_engine_check rime-static)
endif()
